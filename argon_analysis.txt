import numpy as np
from scipy.optimize import minimize

def verify_portfolio_tranche_optimization():
    r0 = 0.050
    mu = np.array([0.64, 0.52, 0.38, 0.32])
    Q = np.diag([0.80, 0.60, 0.50, 0.40])
    a_lev = np.array([1.0, 1.0, 1.0, 0.0])
    b_lev = 0.95
    a_dd = np.array([1.0, 0.25, 0.15, 0.20])
    b_dd = 0.65
    bounds = [(0.15, 0.50), (0.00, 0.30), (0.00, 0.25), (0.00, 0.30)]

    def log_growth(x):
        x = np.asarray(x, dtype=float)
        return float(r0 + np.dot(mu, x) - 0.5 * np.dot(x, Q @ x))

    # Track A: Analytical KKT Optimum
    x_kkt = np.array([0.45, 0.30, 0.20, 0.30])
    lam1_star = 0.28
    nu_star = np.array([0.00, 0.06, 0.00, 0.20])
    g_kkt = log_growth(x_kkt)
    cagr_kkt = np.expm1(g_kkt)

    # Track B: Multi-Start SLSQP Solver
    constraints = [
        {"type": "ineq", "fun": lambda x: b_lev - np.dot(a_lev, x)},
        {"type": "ineq", "fun": lambda x: b_dd - np.dot(a_dd, x)},
    ]
    rng = np.random.default_rng(42)
    best = None
    for _ in range(40):
        x0 = np.array([rng.uniform(lo, hi) for lo, hi in bounds])
        res = minimize(lambda x: -log_growth(x), x0=x0, jac=lambda x: -(mu - Q @ x),
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"ftol": 1e-12, "maxiter": 500})
        if best is None or res.fun < best.fun:
            best = res

    assert np.allclose(x_kkt, best.x, atol=1e-7)
    # Test 1: Feasibility
    assert max(0.0, np.dot(a_lev, x_kkt) - b_lev, np.dot(a_dd, x_kkt) - b_dd) <= 1e-6
    # Test 2: KKT Stationarity & Curvature
    grad_L = (mu - Q @ x_kkt) - lam1_star * a_lev - nu_star
    assert np.max(np.abs(grad_L)) <= 1e-6
    assert np.all(np.linalg.eigvalsh(-Q) < -0.35)
    # Test 3: Perturbation along active x1 + x3 = 0.65 manifold
    for sign in (+1.0, -1.0):

    # Test 4: Baseline comparison vs current config.py
    x_curr = np.array([0.26, 0.0, 0.0, 0.0])
    assert 0.60 <= cagr_kkt <= 0.80 and cagr_kkt > np.expm1(log_growth(x_curr)) + 0.45
    print(f"Verified x* = {x_kkt} -> CAGR* = {cagr_kkt*100:.2f}% (Baseline: {np.expm1(log_growth(x_curr))*100:.2f}%)")
